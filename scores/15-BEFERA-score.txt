# LAB 15 / BEFERA — filtering and synthesis on the organ. Candidate lab score.
# After Befera & Guidarini (2020) on Verrando (Triptych / Filtering) and Romitelli
# (An Index of Metals / Hellucination I). The organ is read as a fixed additive synthesiser.
#
# PART 1 · FILTERING (Verrando). Five timbral categories, each one registration (auditory stream):
#   AP aperiodic   32-foot pedal noise          S  sinusoidal   one stopped flute, almost a sine
#   P  inharmonic  mutations and mixtures       R  inharmonic 2  reed cluster
#   I  impulse     30 ms hits
# Blocks are juxtaposed and repeated: ap · s+p+i · s+r · s+p+i · r · s+r · r · s+p+i. Their lengths
# keep the paper's proportions in quavers (13 20 4 12 4 4 3 16), here 1 quaver = 1.5 s.
# Registration changes stay inside one ten: every block is a neighbouring button.
#
# PART 2 · SYNTHESIS (Romitelli). chord FM:D1:1:M:4 writes the side bands of a symbolic frequency
# modulation on D1. The modulator ratio descends 1:32 → 1:2 through twelve modules whose lengths
# mirror around the middle (the cycle should not be heard), while density and CRESC rise.
id apomixis.lab.befera
version 0.1.0-lab.1
title "BEFERA"
number lab15
duration 284
seed 3151
audition-transpose 0

voice $SO = so range 36 84
voice $RE = re range 36 84
voice $PO = po range 36 84
voice $GO = go range 36 84
voice $PE = pe range 24 55
fertility $SO 0 end
fertility $RE 0 end
fertility $PO 0 end
fertility $GO 0 end
fertility $PE 0 end
harmonicity 0 "resonance" "spectral categories"
harmonicity 284 "resonance" ""

set SINE = chord A4:0 | E5:0 | A4:0 | C#5:0 pat x............... rate 1/16 ms 2600 hold 1 div re
set INHARM = chord C4:0,1,6,11,13,18 | C4:0,2,7,11,14,19 pat x....x..x....... rate 1/16 ms 700 hold 2 morph 1 iris 1 div go,po
set REEDS = chord E2:0,1,2,3,5,6 pat x....... rate 1/8 ms 1400 hold 1 iris 1 div re,po
set IMPULSE = chord C5:0,6,13,19 pat x..x.....x...... rate 1/32 ms 30 hold 3 oct 0|1|-1 div so
set NOISE = pedal ostinato 24 26 25 27 28 rate 1/32 dur 6 rdur 40

section F1 "ap" for 19.5 bpm 80
native off
pgm @noise32
laser off
@NOISE
bio "block 1 · aperiodic: 32-foot noise"
end

section F2 "s+p+i" for 30 bpm 80
native off
pgm III.corDeNuit8 @mutations @mixtures 73
laser off
exp3 0.2~0.8 5
@SINE
@INHARM
@IMPULSE
bio "block 2 · sinusoid + inharmonic + impulses · the box undulates the sine"
end

section F3 "s+r" for 6 bpm 80
native off
pgm III.corDeNuit8 @reeds
laser off
@SINE
@REEDS
bio "block 3 · sinusoid + reed inharmonic"
end

section F4 "s+p+i" for 18 bpm 80
native off
pgm III.corDeNuit8 @mutations @mixtures 73
laser off
exp3 0.8~0.2 5
@SINE
@INHARM
@IMPULSE
bio "block 4 = block 2"
end

section F5 "r" for 6 bpm 80
native off
pgm @reeds @trem
laser off
@REEDS
bio "block 5 · reeds with tremulant"
end

section F6 "s+r" for 6 bpm 80
native off
pgm III.corDeNuit8 @reeds
laser off
@SINE
@REEDS
bio "block 6 = block 3"
end

section F7 "r" for 4.5 bpm 80
native off
pgm @reeds @trem
laser off
@REEDS

end

section F8 "s+p+i" for 24 bpm 80
native off

laser off

@SINE
@INHARM
@IMPULSE
bio "block 8 = block 2 · the sine closes the first part"
end
