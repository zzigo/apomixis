# LAB 17 / LEGACY — the 2020 SuperCollider sketches, line for line. Candidate lab score.
# Source: archive/supercollider/legacy/APOMIXIS GENERATOR.scd. Each Pbind becomes one `bind`:
#   Pseq → a|b|c     Prand → a?b?c     Pshuf → a&b&c     \degree \scale \octave \mtranspose \dur \legato keep their names
#   (mtranspose is written mtr; \chan 0 1 2 3 4 become div go po re so pe).
# The L-system block becomes a header line: lsys NAME = AXIOM iter N rules … map …
# 1 beat = 1 s as in SC's default clock (bpm 60), so durations are the 2020 values:
# 0.15 s per note in the canon, 1/8 s in the L-system — the timings that worked on the organ.
# Not carried over: m.program (Program Change registration), the SimpleMIDIFile export, and
# the out-of-range "+48" voice, here an octave above instead.
id apomixis.lab.legacy
version 0.1.0-lab.1
title "LEGACY 2020"
number lab17
duration 240
seed 2020
audition-transpose 0

voice $SO = so range 36 84
voice $RE = re range 36 84
voice $PO = po range 36 84
voice $GO = go range 36 84
voice $PE = pe range 24 55
fertility $SO 0 end
fertility $RE 0 end
fertility $PO 0 end
fertility $GO 0 end
fertility $PE 0 end
harmonicity 0 "P5" "2020"
harmonicity 240 "P5" ""

# Pseq([ Pshuf(#[-7,-3,0,2,4,7], 4), Pseq([0,1,2,3,4,5,6,7]) ], 1)
set SHUF = -7&-3&0&2&4&7
set CANON = -7|0|2|-3|7|4|-7|0|2|-3|7|4|-7|0|2|-3|7|4|-7|0|2|-3|7|4|0|1|2|3|4|5|6|7
# var dict = [A -> "AB", B -> "A", C -> "DB", D -> "BC"]; word = "AC"; iter = 10
lsys LNOTES = AC iter 10 rules A>AB,B>A,C>DB,D>BC map A:50,B:55,C:54,D:57
lsys LHALF = AC iter 10 rules A>AB,B>A,C>DB,D>BC map A:25,B:27.5,C:27,D:28.5

section A "Organ generator" for 80 bpm 60
native off
pgm 20 23 21 22 61
laser off
bind degree -19|-15|-12|-10|-8|-5|-12|-11|-10|-8|-7|-6|-5|-4 scale 0,1,3,5,7,12 octave 5 dur 2?2?4 legato 1 div go?so?po
bind degree @SHUF scale 0,1,3,5,7,12 octave 4 dur 2?2?4 legato 1 div re
bio "the first Pbind · slow modal line thrown between I, IV and II"
end

section B "Modal transposition" for 80 bpm 60
native off
pgm 20 23 21 22 61
laser off
bind degree @CANON mtr 0 dur 0.15 legato 0.8 div go
bind degree @CANON mtr 1 dur 0.15 legato 0.8 div po
bind degree @CANON mtr 2 dur 0.15 legato 0.8 div re
bio "three Pbinds with mtranspose 0 1 2 · here simultaneous instead of successive: a modal canon at 150 ms"
end

section C "L-system" for 80 bpm 60
native off
pgm 20 23 21 22 61
laser midixy src po size 0 prob 30 max 2 dur 0.08
bind note @LHALF dur 0.125 legato 0.8 div po
bind note @LNOTES tr -24 dur 0.125 legato 0.8 div re
bind note @LHALF dur 0.25 legato 0.8 div go
bio "AC rewritten ten times · the same word at two speeds and two registers"
end
