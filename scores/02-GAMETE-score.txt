# Apomixis 02 / GAMETE — Gameto / el grano. Candidate score from lab 10 IRIS. Seconds; MIDI C4=60.
# Genre: iridescent hocket-chorale. One object (the gamete) seen from many angles; it never fuses.
# OBJ  the object: two open layers a tritone apart (B–F# below, F–C–G above) around the minor second
#      B3–C4 at the centre. It stays the same; only its angle (manual, octave, moment) changes.
# Drift chords: B Phrygian (melancholy, C natural) against F Lydian (B natural). The two modes differ
# by one semitone (F / F#): micro; their roots are a tritone apart: macro.
# Roles: drift chords on I (3+3+2: X..x.x..), their complement on II (.xx.x.xx): together every
# sixteenth sounds, no gaps. Satellites: the object in running hockets at double speed (1/32, odd
# thirty-seconds only) on Manual III, never on the same tick as I–II: encounters that do not arrive.
# Glints: hardcore techno at 150 BPM. 32' noise kick on every beat, diffuse FM clusters in 1/32,
# the object as off-beat stab; white full-field laser flash on kick and stab.
# The end turns to major chords (the Fell side), the Interference pedal comes back, and Manual IV is
# kept silent for the super brass (Tuba magna 16, Tuba 8), heard only in Glint 4 and the re-exposition.
# Programs 20–27 (eight, inside the ten of 02). Previous tonal draft: archive/scores/gamete/.

# The object (fixed).
set OBJ = B2,F#3,B3,C4,F4,C5,G5
# B Phrygian angles: B minor with G above (bVI), E–B open fifth with G–A suspended.
set PH1 = B2,F#3,B3,C4,G4,D5
set PH2 = E3,B3,C4,G4,A4,E5
# F Lydian angles: F–C–G fifths with B (#4); A–E open with F–G–D above.
set LY1 = F2,C3,G3,B3,C4,E4,A4
set LY2 = A2,E3,B3,C4,F4,G4,D5
# Polytonal: Bb–F and B–F#–C# a semitone apart, the centre still B–C.
set POLY = Bb2,F3,B3,C4,F#4,C#5
# Major chords for the end: Bmaj9, Gmaj9, Fmaj7#11 and the IRIS film (Emaj9).
set MJ1 = B2,F#3,C#4,D#4,A#4,F#5
set MJ2 = G2,D3,A3,B3,F#4,D5
set MJ3 = F2,C3,A3,E4,B4,G5
set FILM = E3:0,7,14,16,21,23
# The Interference pedal of IRIS, unchanged.
set DEEP = D3:0,7,14,17,21,26
# Registrations.
